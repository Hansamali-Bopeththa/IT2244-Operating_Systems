IPC using Message Queue

read inputs from the parent process
Enter name: hi
Enter RegNo: 2314
Enter Age: 24

Message sent successfully.

give the output from thechild prosess

Received Name: hi
Received RegNO: 2314
Received Age: 24



(I) send

#include <stdio.h>
#include <sys/ipc.h>
#include <sys/msg.h>

#define MAX 100

// Structure for message queue
struct mesg_buffer {
    long mesg_type;
    char name[50];
    int regNo;
    int age;
} message;

int main() {
    key_t key;
    int msgid;

    // Generate unique key
    key = ftok("2021ict35", 65);

    // Create message queue
    msgid = msgget(key, 0666 | IPC_CREAT);
    message.mesg_type = 1;

    // Read input from user
    printf("Enter Name: ");
    scanf("%s", message.name);
    printf("Enter RegNo: ");
    scanf("%d", &message.regNo);
    printf("Enter Age: ");
    scanf("%d", &message.age);

    // Send message
    msgsnd(msgid, &message, sizeof(message) - sizeof(long), 0);

    // Confirmation of message sent
    printf("Message sent successfully.\n");

    return 0;
}


(II) receive
#include <stdio.h>
#include <sys/ipc.h>
#include <sys/msg.h>

// Structure for message queue
struct mesg_buffer {
    long mesg_type;
    char name[50];
    int regNo;
    int age;
} message;

int main() {
    key_t key;
    int msgid;

    // Generate unique key
    key = ftok("2021ict35", 65);

    // Access message queue
    msgid = msgget(key, 0666 | IPC_CREAT);

    // Receive message
    msgrcv(msgid, &message, sizeof(message) - sizeof(long), 1, 0);

    // Display received data
    printf("Received Name: %s\n", message.name);
    printf("Received RegNo: %d\n", message.regNo);
    printf("Received Age: %d\n", message.age);

    // Destroy message queue
    msgctl(msgid, IPC_RMID, NULL);

    return 0;
}
