Q7.)
Create simple calculator using "expr" command same as the question 3


echo "Enter the number 1:"
read num1

echo "Enter the number 2:"
read num2

sum=$(expr $num1 + $num2)
echo "Summation: $sum"

sub=$(expr $num1 - $num2)
echo "Subtraction: $sub"

mul=$(expr $num1 \* $num2)
echo "Multiplication: $mul"


if [ $num2 -ne 0 ]; then
    div=$(expr $num1 / $num2)
    echo "Division: $div"
else
   echo "Division: Error! Cannot divide by zero."
fi
